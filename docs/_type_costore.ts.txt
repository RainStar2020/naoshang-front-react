import type { RichContent } from './cms'
export type CostoreApplicationStatus = 'PENDING_PAYMENT' | 'PENDING_REVIEW' | 'APPROVED' | 'REJECTED' | 'CANCELLED'
export type CostorePaymentStatus = 'NOT_STARTED' | 'PENDING' | 'PAID' | 'CANCELLED'
export type CostoreRefundStatus = 'PENDING' | 'SUCCEEDED' | 'FAILED' | null
export interface CostorePlan { planId: string; title: string; durationId: string; months: number; priceBoundary: { referenceMonthlyFee: number; note: string }; benefits: RichContent; agreementVersion: number | null; agreement: RichContent; version: number }
export interface CostoreApplication { applicationId: string; accountId: string; applicantName: string; applicantPhone: string; storeId: string; storeName: string; durationId: string; durationLabel: string; monthlyFee: number; payableAmount: number; orderId: string | null; applicationStatus: CostoreApplicationStatus; paymentStatus: CostorePaymentStatus; refundStatus: CostoreRefundStatus; refundId: string | null; contractStatus: 'NOT_SIGNED' | 'PENDING_CONFIRMATION' | 'SIGNED'; entitlementStatus: 'NOT_OPEN' | 'PENDING_CONFIRMATION' | 'OPEN'; agreementVersion: number | null; createdAt: string; paidAt: string | null; reviewedAt: string | null; rejectionReason: string | null; version: number; allowedActions: ('retry-refund')[] }
export interface CostoreRefund { refundId: string; applicationId: string; orderId: string; amount: number; status: Exclude<CostoreRefundStatus, null>; paymentReceiptId: string; reason: string; createdAt: string; completedAt: string | null; failureReason: string | null; mockOnly: true }
export interface CostoreAudit { operationId: string; requestId: string; action: 'save-plan' | 'retry-refund'; targetId: string; operatorName: string; reason: string; before: string; after: string; occurredAt: string; beforeVersion: number; afterVersion: number }
export interface CostoreCatalog { plans: CostorePlan[]; stores: { storeId: string; name: string; managerName: string; ownerCount: number; active: boolean }[]; refundRetryAllowed: boolean; mockOnly: true }
export interface CostoreQuery { page: number; pageSize: number; keyword?: string; applicationStatus?: CostoreApplicationStatus; paymentStatus?: CostorePaymentStatus; refundStatus?: Exclude<CostoreRefundStatus, null> }
export interface CostoreList { items: CostoreApplication[]; total: number; summary: { applications: number; paid: number; pendingReview: number; rejected: number; refundPending: number; refundFailed: number } }
export interface CostoreDetail { application: CostoreApplication; refunds: CostoreRefund[]; audit: CostoreAudit[] }
export type CostoreMutation = { reason: string; idempotencyKey: string; version: number } & ({ action: 'save-plan'; planId: string; title: string; benefits: RichContent; agreement: RichContent } | { action: 'retry-refund'; applicationId: string })
export class CostoreError extends Error { constructor(public code: 'FORBIDDEN' | 'NOT_FOUND' | 'INVALID_INPUT' | 'INVALID_STATE' | 'VERSION_CONFLICT' | 'IDEMPOTENCY_CONFLICT', message: string) { super(message); this.name = 'CostoreError' } }
